github.yaml
# REAL source: GitHub REST API. Login: Bearer token (fine-grained PAT, read-only). Paging: Link header with page numbers.
name: github
display_name: "GitHub"
enabled: true
base_url: "https://api.github.com"
auth: { type: bearer, token: "${github_token}" }
scopes: ["metadata:read", "dependabot_alerts:read"]   # fine-grained PAT permissions (read-only)
headers:
  Accept: "application/vnd.github+json"
  X-GitHub-Api-Version: "2022-11-28"
schedule_seconds: 900
rate_limit: { reset_header: "x-ratelimit-reset", remaining_header: "x-ratelimit-remaining", min_remaining: 1 }
streams:
  - name: me
    entity: user
    endpoint: "/user"
    mapping: { id: "id", name: "name || login", email: "email", status: "'active'" }
  - name: collaborators
    entity: user
    endpoint: "/repos/${GITHUB_OWNER}/${GITHUB_REPO}/collaborators"
    params: { per_page: 100 }
    pagination: { type: link, max_pages: 10 }
    mapping: { id: "id", name: "login", email: "email", status: "role_name" }
  - name: org_members          # only when GITHUB_ORG is set
    entity: user
    requires_env: ["GITHUB_ORG"]
    endpoint: "/orgs/${GITHUB_ORG}/members"
    params: { per_page: 100 }
    pagination: { type: link, max_pages: 10 }
    mapping: { id: "id", name: "login", status: "'member'" }
  - name: dependabot_alerts
    entity: event
    endpoint: "/repos/${GITHUB_OWNER}/${GITHUB_REPO}/dependabot/alerts"
    params: { per_page: 100, state: "open" }
    pagination: { type: link, max_pages: 10 }
    mapping:
      id: "join('-', ['dependabot', to_string(number)])"
      type: "'dependabot_alert'"
      severity: "security_advisory.severity"
      timestamp: "updated_at"
      description: "join('', [security_advisory.summary, ' (', dependency.package.name, ')'])"
  - name: repo_events          # used only when Dependabot alerts are not available (HTTP 403/404)
    entity: event
    fallback_for: dependabot_alerts
    endpoint: "/repos/${GITHUB_OWNER}/${GITHUB_REPO}/events"
    params: { per_page: 100 }
    pagination: { type: link, max_pages: 3 }
    mapping:
      id: "id"
      type: "type"
      timestamp: "created_at"
      description: "join(' ', [actor.login, type, repo.name])"
      user: "actor.login"

okta.yaml
# REAL source: Okta Integrator Free Plan. Login: API token header ("SSWS <token>"). Paging: Link header with an `after` cursor.
name: okta
display_name: "Okta"
enabled: true
base_url: "https://${OKTA_DOMAIN}"
auth: { type: api_key, header: "Authorization", prefix: "SSWS ", api_key: "${okta_token}" }
scopes: ["okta.users.read", "okta.logs.read"]   # create the token as a read-only admin
schedule_seconds: 900
rate_limit: { reset_header: "x-rate-limit-reset", remaining_header: "x-rate-limit-remaining", min_remaining: 2 }
streams:
  - name: users
    entity: user
    endpoint: "/api/v1/users"
    params: { limit: 200 }
    pagination: { type: link, max_pages: 20 }
    mapping:
      id: "id"
      name: "join(' ', [profile.firstName || '', profile.lastName || ''])"
      email: "profile.email"
      status: "status"
      last_login: "lastLogin"
  - name: events
    entity: event
    endpoint: "/api/v1/logs"
    params: { limit: 1000, since: "{{ last_sync }}" }   # incremental (B-1); first run uses Okta's default window
    pagination: { type: link, max_pages: 5 }
    mapping:
      id: "uuid"
      type: "eventType"
      severity: "severity"
      timestamp: "published"
      description: "displayMessage"
      user: "actor.alternateId"
      src_ip: "client.ipAddress"
    transforms:
      severity_map: { DEBUG: info, INFO: info, WARN: medium, ERROR: high }

jira.yaml
# REAL source: Jira Cloud (free plan). Login: Basic (email + API token). Paging: offset (startAt / maxResults).
name: jira
display_name: "Jira Cloud"
enabled: true
base_url: "https://${JIRA_DOMAIN}"
auth: { type: basic, username: "${jira_email}", password: "${jira_token}" }
scopes: ["read:jira-user", "read:jira-work"]
schedule_seconds: 900
streams:
  - name: users
    entity: user
    endpoint: "/rest/api/3/users/search"
    pagination: { type: offset, offset_param: startAt, limit_param: maxResults, limit: 50, max_pages: 20 }
    mapping:
      id: "accountId"
      name: "displayName"
      email: "emailAddress"
      status: "active && 'active' || 'inactive'"
  - name: issues               # the classic /search endpoint is deprecated: /search/jql pages with nextPageToken
    entity: event
    endpoint: "/rest/api/3/search/jql"
    params:
      jql: "updated >= -30d ORDER BY updated DESC"
      fields: "summary,priority,issuetype,updated,status"
      maxResults: 50
    pagination: { type: cursor, cursor_path: "nextPageToken", cursor_param: "nextPageToken", max_pages: 5 }
    records_path: "issues"
    mapping:
      id: "key"
      type: "fields.issuetype.name"
      severity: "fields.priority.name"
      timestamp: "fields.updated"
      description: "fields.summary"
    transforms:
      severity_map: { Highest: critical, High: high, Medium: medium, Low: low, Lowest: info }

nvd.yaml
# REAL source: NVD CVE API 2.0. Login: optional API key header (works without one, at a lower rate limit).
# Paging: offset via startIndex / resultsPerPage / totalResults. Incremental: lastModStartDate..lastModEndDate (max 120 days).
name: nvd
display_name: "NVD CVE API 2.0"
enabled: true
base_url: "https://services.nvd.nist.gov"
auth: { type: api_key, header: "apiKey", api_key: "${nvd_api_key}", optional: true }
schedule_seconds: 900
retry: { max_retries: 3, backoff: exponential, base_seconds: 1 }
# NVD without a key: 5 requests / rolling 30 s. Pace at 6 s; if a 429 still arrives (it says "Retry-After: 0"), wait out the window.
rate_limit: { min_interval_seconds: 6, fallback_wait_seconds: 30 }
exfiltration_guard: { enabled: false }   # public data, no token to abuse; NVD's daily bulk updates would be false positives
streams:
  - name: cves
    entity: event
    endpoint: "/rest/json/cves/2.0"
    params:
      lastModStartDate: "{{ (last_sync_dt or (now_dt - days(1))).strftime('%Y-%m-%dT%H:%M:%S.000Z') }}"
      lastModEndDate: "{{ now_dt.strftime('%Y-%m-%dT%H:%M:%S.000Z') }}"
    pagination: { type: offset, offset_param: startIndex, limit_param: resultsPerPage, limit: 1000, total_path: totalResults, max_pages: 5 }
    records_path: "vulnerabilities"
    timeout_seconds: 60
    mapping:
      id: "cve.id"
      type: "'vulnerability'"
      severity: "cve.metrics.cvssMetricV31[0].cvssData.baseSeverity || cve.metrics.cvssMetricV30[0].cvssData.baseSeverity || cve.metrics.cvssMetricV2[0].baseSeverity"
      timestamp: "cve.lastModified"
      description: "cve.descriptions[?lang=='en'] | [0].value"
    transforms:
      severity_map: { LOW: low, MEDIUM: medium, HIGH: high, CRITICAL: critical }
      default_severity: info

    Built to fail safely

    Every connector handles the same four failures the same way. You can trigger each one from the dashboard.

    Wrong key

    401, shown as "Auth failed". No retry: a wrong key never gets better.

    Rate limit

    429 waits exactly as long as Retry-After says, then carries on.

    Network off

    Retries after 1 s, 2 s and 4 s, then the run is marked failed with a clear error.

    Bad record

    Quarantined with its reason. The good records are still stored.

    One format for every tool

    Okta logs, GitHub alerts, Jira issues and CVE feeds all arrive as the same event: severity, type, time, who and where.
